"""
Cycle Scanner
A working implementation of the four-step framework from the FSC whitepaper.
Dependencies: numpy, scipy.
"""

import numpy as np
from scipy.sparse import eye, diags
from scipy.sparse.linalg import spsolve


# ----------------------------------------------------------------------
# Step 1 Detrending (Hodrick-Prescott filter)
# ----------------------------------------------------------------------
def hp_filter(y, lam=1600.0) -> tuple[np.ndarray, np.ndarray]:
    """Separate data into a trend and a cycle. Higher lambda values create a smoother trend."""
    y = np.asarray(y, dtype=float)
    n = len(y)
    if n < 3:
        return y.copy(), np.zeros_like(y)

    # second-difference matrix D (size (n-2) x n)
    diagonals = [np.ones(n), -2.0 * np.ones(n), np.ones(n)]
    D = diags(diagonals, [0, 1, 2], shape=(n - 2, n))

    I = eye(n)
    trend = spsolve((I + lam * (D.T @ D)).tocsc(), y)
    cycle = y - trend
    return trend, cycle


# ----------------------------------------------------------------------
# Step 2 Cycle detection (Goertzel)
# ----------------------------------------------------------------------
def goertzel(x, period) -> tuple[float, float]:
    """Calculate the amplitude and phase of a specific cycle period. The phase starts at the first sample."""
    x = np.asarray(x, dtype=float)
    n = len(x)
    omega = 2.0 * np.pi / period
    cos_w = np.cos(omega)
    sin_w = np.sin(omega)
    coeff = 2.0 * cos_w

    s_prev = 0.0
    s_prev2 = 0.0
    for sample in x:
        s = sample + coeff * s_prev - s_prev2
        s_prev2 = s_prev
        s_prev = s

    real = s_prev - s_prev2 * cos_w
    imag = s_prev2 * sin_w

    amplitude = 2.0 * np.hypot(real, imag) / n
    phase = np.arctan2(imag, real)
    return amplitude, phase


def scan_periods(x, min_period=8, max_period=None, step=1.0) -> list[tuple[float, float]]:
    """Scan a range of periods using the Goertzel algorithm. Return results sorted by highest amplitude."""
    n = len(x)
    if max_period is None:
        max_period = n // 2  # cannot trust cycles longer than half the data

    results = []
    p = float(min_period)
    while p <= max_period:
        amp, _ = goertzel(x, p)
        results.append((p, amp))
        p += step

    results.sort(key=lambda r: r[1], reverse=True)
    return results


def fit_sinusoid(x, period) -> tuple[float, float, float]:
    """Perform a least-squares fit for a single sinusoid. This provides clean amplitude and phase data for projections."""
    x = np.asarray(x, dtype=float)
    n = len(x)
    t = np.arange(n)
    omega = 2.0 * np.pi / period

    # model: x ~ a*cos(omega t) + b*sin(omega t) + c
    A = np.column_stack([np.cos(omega * t), np.sin(omega * t), np.ones(n)])
    coef, *_ = np.linalg.lstsq(A, x, rcond=None)
    a, b, _ = coef

    amplitude = np.hypot(a, b)
    phase_first = np.arctan2(-b, a)               # phase at t = 0
    phase_last = (phase_first + omega * (n - 1))  # phase at the last bar
    phase_last = (phase_last + np.pi) % (2 * np.pi) - np.pi
    return amplitude, phase_first, phase_last


# ----------------------------------------------------------------------
# Step 3 Validation (Bartels test)
# ----------------------------------------------------------------------
def bartels_test(x, period) -> tuple[float, float]:
    """Test the statistical significance of a cycle. Return the probability score and the genuine percentage."""
    x = np.asarray(x, dtype=float)
    P = int(round(period))
    n = len(x)
    k = n // P  # number of whole cycles available
    if k < 2 or P < 2:
        return 1.0, 0.0  # not enough data to judge, treat as random

    t = np.arange(P)
    cos_t = np.cos(2.0 * np.pi * t / P)
    sin_t = np.sin(2.0 * np.pi * t / P)

    phasors = []
    for j in range(k):
        seg = x[j * P:(j + 1) * P]
        real = np.dot(seg, cos_t)
        imag = -np.dot(seg, sin_t)
        phasors.append(complex(real, imag))

    phasors = np.array(phasors)
    resultant = np.abs(np.sum(phasors))           # length of the vector sum
    sum_sq = np.sum(np.abs(phasors) ** 2)         # expected spread if random
    if sum_sq == 0:
        return 1.0, 0.0

    bartels_score = float(np.exp(-(resultant ** 2) / sum_sq))
    genuine_pct = (1.0 - bartels_score) * 100.0
    return bartels_score, genuine_pct


# ----------------------------------------------------------------------
# Step 4 Ranking + full pipeline
# ----------------------------------------------------------------------
def cycle_scan(y, lam=1600.0, min_period=8, max_period=None, step=1.0, top_candidates=25, genuine_threshold=49.0) -> list[dict]:
    """Run the complete four-step cycle detection pipeline. Return valid cycles sorted by strength per bar."""
    _, cyc = hp_filter(y, lam=lam)

    ranked = scan_periods(cyc, min_period, max_period, step)
    candidates = ranked[:top_candidates]

    out = []
    for period, _scan_amp in candidates:
        amplitude, phase_first, phase_last = fit_sinusoid(cyc, period)
        bartels_score, genuine = bartels_test(cyc, period)
        if genuine < genuine_threshold:
            continue  # drop cycles that fail validation
        out.append({
            "period": round(period, 2),
            "amplitude": round(amplitude, 4),
            "strength_per_bar": round(amplitude / period, 5),
            "genuine_pct": round(genuine, 1),
            "bartels_score": round(bartels_score, 4),
            "phase_last_bar_deg": round(np.degrees(phase_last), 1),
        })

    out.sort(key=lambda c: c["strength_per_bar"], reverse=True)
    return out


def project_cycle(cycle_info, history_len, bars_ahead=30) -> np.ndarray:
    """Rebuild a valid cycle and project it into the future. This generates future values for charting."""
    period = cycle_info["period"]
    amp = cycle_info["amplitude"]
    phase_last = np.radians(cycle_info["phase_last_bar_deg"])
    omega = 2.0 * np.pi / period
    future_t = np.arange(1, bars_ahead + 1)
    return amp * np.cos(omega * future_t + phase_last)


# ----------------------------------------------------------------------
# Demo plant known cycles in fake data and check we recover them
# ----------------------------------------------------------------------
def _demo() -> tuple[np.ndarray, list[dict]]:
    """Generate fake data with known cycles to test the scanner. Print the results to the console."""
    rng = np.random.default_rng(42)
    n = 600
    t = np.arange(n)

    # three planted cycles, a trend, and noise
    signal = (
        1.0 * np.cos(2 * np.pi * t / 20 + 0.5) +     # period 20
        0.6 * np.cos(2 * np.pi * t / 33 + 1.2) +     # period 33
        0.4 * np.cos(2 * np.pi * t / 55 + 2.0)       # period 55
    )
    trend = 0.004 * t + 1e-5 * (t - n / 2) ** 2
    noise = rng.normal(0, 0.7, n)                     # heavy noise
    y = signal + trend + noise

    print("Planted cycles: period 20 (amp 1.0), 33 (amp 0.6), 55 (amp 0.4)")
    print("Trend and heavy noise added.\n")

    results = cycle_scan(y, lam=1600, min_period=8, max_period=120, step=1.0)

    print(f"{'period':>7} {'amp':>7} {'strength/bar':>13} "
          f"{'genuine%':>9} {'phase_last':>11}")
    print("-" * 52)
    for c in results[:8]:
        print(f"{c['period']:>7} {c['amplitude']:>7} "
              f"{c['strength_per_bar']:>13} {c['genuine_pct']:>8}% "
              f"{c['phase_last_bar_deg']:>9}")

    if results:
        dom = results[0]
        print(f"\nDominant cycle: period {dom['period']} bars, "
              f"genuine {dom['genuine_pct']}%")
        fwd = project_cycle(dom, n, bars_ahead=10)
        print("Next 10 projected values of the dominant cycle:")
        print(np.round(fwd, 3))

    return y, results


if __name__ == "__main__":
    _demo()

import numpy as np
import matplotlib.pyplot as plt

def plot_composite_wave(y_detrended, valid_cycles):
    n = len(y_detrended)
    t_hist = np.arange(n)
    t_rev = np.arange(-n + 1, 1)
    
    composite = np.zeros(n)
    
    for c in valid_cycles:
        omega = 2.0 * np.pi / c["period"]
        phase = np.radians(c["phase_last_bar_deg"])
        wave = c["amplitude"] * np.cos(omega * t_rev + phase)
        composite += wave
        
    plt.figure(figsize=(10, 4))
    plt.plot(t_hist, y_detrended, color="lightgray", label="Detrended Price")
    plt.plot(t_hist, composite, color="blue", label="Composite Model")
    plt.title("Real Data vs Composite Cycle Model")
    plt.legend()
    plt.grid(alpha=0.3)
    plt.tight_layout()
    plt.show()

def plot_validation_scatter(valid_cycles):
    periods = [c["period"] for c in valid_cycles]
    scores = [c["genuine_pct"] for c in valid_cycles]
    
    plt.figure(figsize=(10, 4))
    plt.scatter(periods, scores, color="black")
    plt.axhline(y=95.0, color="red", linestyle="--", label="95% Confidence")
    plt.title("Cycle Period vs Genuine Percentage")
    plt.xlabel("Period")
    plt.ylabel("Genuine %")
    plt.legend()
    plt.grid(alpha=0.3)
    plt.tight_layout()
    plt.show()



import numpy as np
import matplotlib.pyplot as plt
from scipy.signal import find_peaks

def get_clean_peaks(scan_results, min_distance=3):
    sorted_scan = sorted(scan_results, key=lambda x: x[0])
    periods = np.array([x[0] for x in sorted_scan])
    amplitudes = np.array([x[1] for x in sorted_scan])
    
    strengths = amplitudes / periods
    
    peak_idx, _ = find_peaks(strengths, distance=min_distance)
    
    peaks = []
    for i in peak_idx:
        peaks.append({
            "period": periods[i],
            "amplitude": amplitudes[i],
            "strength_per_bar": strengths[i]
        })
        
    peaks.sort(key=lambda c: c["strength_per_bar"], reverse=True)
    return peaks

def plot_single_series(y_data, title_text, y_label_text):
    plt.figure(figsize=(10, 4))
    plt.plot(y_data, color="black")
    plt.title(title_text)
    plt.ylabel(y_label_text)
    plt.grid(alpha=0.3)
    plt.tight_layout()
    plt.show()

def plot_spectrum(scan_results):
    sorted_scan = sorted(scan_results, key=lambda x: x[0])
    periods = [x[0] for x in sorted_scan]
    strengths = [x[1] / x[0] for x in sorted_scan]
    
    plt.figure(figsize=(10, 4))
    plt.plot(periods, strengths, color="black")
    plt.title("Cycle Spectrum")
    plt.xlabel("Period")
    plt.ylabel("Strength per Bar")
    plt.grid(alpha=0.3)
    plt.tight_layout()
    plt.show()
